    <script>
        // QR Code generation using a simple algorithm
        const qrContent = document.getElementById('qrContent');
        const fgColor = document.getElementById('fgColor');
        const bgColor = document.getElementById('bgColor');
        const qrSize = document.getElementById('qrSize');
        const generateBtn = document.getElementById('generateBtn');
        const downloadBtn = document.getElementById('downloadBtn');
        const qrCanvas = document.getElementById('qrCanvas');
        const errorMsg = document.getElementById('errorMsg');
        const fgValue = document.getElementById('fgValue');
        const bgValue = document.getElementById('bgValue');

        // Update color value displays
        fgColor.addEventListener('input', () => {
            fgValue.textContent = fgColor.value.toUpperCase();
        });

        bgColor.addEventListener('input', () => {
            bgValue.textContent = bgColor.value.toUpperCase();
        });

        // Generate QR Code
        generateBtn.addEventListener('click', () => {
            const content = qrContent.value.trim();
            
            if (!content) {
                errorMsg.classList.add('show');
                qrCanvas.classList.remove('show');
                downloadBtn.classList.remove('show');
                return;
            }

            errorMsg.classList.remove('show');
            generateQRCode(content);
        });

        // Download QR Code
        downloadBtn.addEventListener('click', () => {
            const link = document.createElement('a');
            link.download = 'qr-code.png';
            link.href = qrCanvas.toDataURL();
            link.click();
        });

        // Generate QR using QR Code API (we'll use a simple visual representation)
        function generateQRCode(text) {
            const size = parseInt(qrSize.value);
            const fg = fgColor.value;
            const bg = bgColor.value;

            // Use Google Charts API as a fallback for QR generation
            const apiUrl = `https://api.qrserver.com/v1/create-qr-code/?size=${size}x${size}&data=${encodeURIComponent(text)}&color=${fg.replace('#', '')}&bgcolor=${bg.replace('#', '')}`;

            const img = new Image();
            img.crossOrigin = 'anonymous';
            
            img.onload = function() {
                qrCanvas.width = size;
                qrCanvas.height = size;
                const ctx = qrCanvas.getContext('2d');
                ctx.drawImage(img, 0, 0, size, size);
                qrCanvas.classList.add('show');
                downloadBtn.classList.add('show');
            };

            img.onerror = function() {
                // Fallback: create a simple visual pattern if API fails
                createFallbackQR(text, size, fg, bg);
            };

            img.src = apiUrl;
        }

        // Fallback QR (simple visual pattern)
        function createFallbackQR(text, size, fg, bg) {
            qrCanvas.width = size;
            qrCanvas.height = size;
            const ctx = qrCanvas.getContext('2d');
            
            // Fill background
            ctx.fillStyle = bg;
            ctx.fillRect(0, 0, size, size);
            
            // Create a simple pattern based on text
            const gridSize = 25;
            const cellSize = size / gridSize;
            
            ctx.fillStyle = fg;
            
            // Generate pattern from text hash
            for (let i = 0; i < gridSize; i++) {
                for (let j = 0; j < gridSize; j++) {
                    const index = (i * gridSize + j) % text.length;
                    const charCode = text.charCodeAt(index);
                    
                    if ((charCode + i + j) % 2 === 0) {
                        ctx.fillRect(j * cellSize, i * cellSize, cellSize, cellSize);
                    }
                }
            }
            
            // Add border
            ctx.strokeStyle = fg;
            ctx.lineWidth = 3;
            ctx.strokeRect(0, 0, size, size);
            
            qrCanvas.classList.add('show');
            downloadBtn.classList.add('show');
        }

        // Generate on Enter key
        qrContent.addEventListener('keypress', (e) => {
            if (e.key === 'Enter') {
                generateBtn.click();
            }
        });
    </script>
